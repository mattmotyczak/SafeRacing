   1. SERVER.JS UPDATES (HANDLING RELATIONS)
   ============================================================== */
// Since your data is split across two tables (easy_questions and easy_answers),
// a single question actually has 4 rows of answers. We will use a SQL JOIN to 
// fetch them all at once, and then group them together in Javascript so 
// React gets the clean { question, options, answer, photoString } format.

// Add this endpoint to your server.js
app.get('/api/questions/easy', async (req, res) => {
  try {
    // We JOIN the questions and answers tables on the relatedToQuestion foreign key
    const query = `
      SELECT 
        q."idEasyQuestion", q.question, q."photoString",
        a."idAnswer", a.answer, a."isCorrect"
      FROM easy_questions q
      JOIN easy_answers a ON q."idEasyQuestion" = a."relatedToQuestion"
    `;
    const result = await pool.query(query);

    // Because of the JOIN, the database returns 4 rows for every 1 question.
    // We use an object map to group those 4 rows back into a single question object.
    const questionsMap = {};

    result.rows.forEach(row => {
      const qId = row.idEasyQuestion;
      
      // If we haven't seen this question yet in our loop, create its base structure
      if (!questionsMap[qId]) {
        questionsMap[qId] = {
          question: row.question,
          photoString: row.photoString, // Adding the base64 photo!
          options: [],
          answer: -1 
        };
      }

      const currentQ = questionsMap[qId];
      
      // The index this answer will have in the options array (0, 1, 2, or 3)
      const optionIndex = currentQ.options.length; 
      
      // Push the answer text into the options array
      currentQ.options.push(row.answer);
      
      // If this specific answer row is marked as correct in the DB, 
      // we record its index so React knows which option is the winner
      if (row.isCorrect) {
        currentQ.answer = optionIndex;
      }
    });

    // Object.values converts our map back into a simple array of question objects
    const formattedQuestions = Object.values(questionsMap);

    // Send the data to React!
    res.json(formattedQuestions);
  } catch (err) {
    console.error('Database error:', err);
    res.status(500).json({ error: 'Failed to fetch questions' });
  }
});
