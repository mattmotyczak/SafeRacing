// same structure it currently uses: { question, options: [...], answer }

// Update your /api/questions endpoint in server.js to this:
app.get('/api/questions', async (req, res) => {
  try {
    const result = await pool.query('SELECT * FROM questions');
    
    // map() loops through every row returned by NeonDB
    const formattedQuestions = result.rows.map(row => {
      return {
        mode: row.mode, // We keep 'easy' or 'hard' so we can split them on the frontend
        question: row.question,
        // We combine the 4 columns into a single array, exactly like your hardcoded arrays
        options: [row.option_1, row.option_2, row.option_3, row.option_4],
        answer: row.answer_index
      };
    });

    // Send the perfectly formatted data back to React
    res.json(formattedQuestions);
  } catch (err) {
    console.error('Database error:', err);
    res.status(500).json({ error: 'Failed to fetch questions' });
  }
});
